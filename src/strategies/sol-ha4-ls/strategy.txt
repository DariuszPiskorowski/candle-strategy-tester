//   - wygładzony Heikin Ashi 4H (sygnał wejścia/wyjścia),
//   - dzienny wygładzony HA z ostatniego zamkniętego dnia (filtr kierunku),
//   - własny egzemplarz detektora boczniaka (wygładzanie, potwierdzenie, min. czas trwania),
//   - własny stop loss (k × ATR).
//  Wskaźniki bazowe detektora (ADX, Choppiness, ER, BBW, nachylenie EMA) są wspólne.
//
//  LONG : pierwsza zamknięta ZIELONA świeca HA 4H + dzienny HA zielony + brak boczniaka.
//         Wyjście: czerwona świeca HA 4H albo boczniak. Stop = wejście - kL × ATR.
//  SHORT: pierwsza zamknięta CZERWONA świeca HA 4H + dzienny HA czerwony + brak boczniaka.
//         Wyjście: zielona świeca HA 4H albo boczniak. Stop = wejście + kS × ATR.
//  Jedno konto: nowa pozycja otwiera się TYLKO gdy konto jest płaskie (brak odwracania).
//  Po stopie strona czeka na nowy sygnał (zmiana koloru jej HA 4H tam i z powrotem).
//  Wielkość pozycji: docelowa ekspozycja × kapitał / cena, zaokrąglona w dół do kroku (domyślnie
//  0,1 SOL), nie więcej niż pozwala depozyt przy dźwigni ustawionej na giełdzie.
//  Sygnały na zamkniętej świecy, zlecenia na otwarciu następnej. Funding nie jest liczony.
// ============================================================================

// ============================================================================
//  Wygładzony Heikin Ashi
// ============================================================================
smoothedHaGreen(simple int l1, simple int l2) =>
    float so = ta.ema(open,  l1)
    float sh = ta.ema(high,  l1)
    float sl = ta.ema(low,   l1)
    float sc = ta.ema(close, l1)
    float hc = (so + sh + sl + sc) / 4
    var float ho = na
    ho := na(ho[1]) ? (so + sc) / 2 : (ho[1] + hc[1]) / 2
    float hoS = ta.ema(ho, l2)
    float hcS = ta.ema(hc, l2)
    hcS > hoS

bool greenL  = smoothedHaGreen(haL1L, haL2L)
bool greenS  = smoothedHaGreen(haL1S, haL2S)
bool dGreenL = request.security(syminfo.tickerid, "D", smoothedHaGreen(dL1L, dL2L)[1], lookahead = barmerge.lookahead_on)
bool dGreenS = request.security(syminfo.tickerid, "D", smoothedHaGreen(dL1S, dL2S)[1], lookahead = barmerge.lookahead_on)

// ============================================================================
//  Detektor boczniaka
// ============================================================================
atrD = ta.atr(atrLen)
[diPlus, diMinus, adx] = ta.dmi(adxLen, adxLen)

trSum = math.sum(ta.tr(true), chopLen)
hhC   = ta.highest(high, chopLen)
llC   = ta.lowest(low, chopLen)
chop  = hhC - llC > 0 ? 100 * math.log10(trSum / (hhC - llC)) / math.log10(chopLen) : na

pathSum = math.sum(math.abs(ta.change(close)), erLen)
er      = pathSum > 0 ? math.abs(close - close[erLen]) / pathSum : na

basis  = ta.sma(close, bbLen)
dev    = ta.stdev(close, bbLen)
bbw    = basis != 0 ? (2 * bbMult * dev) / basis : na
bbwPct = ta.percentrank(bbw, bbwLookback) / 100

emaSl    = ta.ema(close, slopeEmaLen)
slopeAtr = atrD > 0 ? (emaSl - emaSl[slopeLen]) / (slopeLen * atrD) : na

bool  validCore = not na(adx) and not na(chop) and not na(er) and not na(slopeAtr)
int   vAdx   = not na(adx)      and adx < adxMax                  ? 1 : 0
int   vChop  = not na(chop)     and chop > chopMin                ? 1 : 0
int   vEr    = not na(er)       and er < erMax                    ? 1 : 0
int   vSlope = not na(slopeAtr) and math.abs(slopeAtr) < slopeMax ? 1 : 0
bool  bbwOk  = useBbw and not na(bbwPct)      // BBW wyłączony -> głos nie liczy się ani w sumie, ani w dzielniku
int   vBbw   = bbwOk and bbwPct < bbwPctMax ? 1 : 0
float rawScore = validCore ? (vAdx + vChop + vEr + vSlope + vBbw) / (bbwOk ? 5.0 : 4.0) : na

hhR = ta.highest(high, rangeLen)[1]
llR = ta.lowest(low, rangeLen)[1]

// Detektor boczniaka - osobny egzemplarz (własny stan) dla każdego wywołania
rangeDetector(simple int smoothLen, simple int confirmBars, simple int minDwell) =>
    float score = na
    if smoothLen == 1
        score := rawScore
    else
        float s  = 0.0
        bool  ok = true
        for i = 0 to smoothLen - 1
            if na(rawScore[i])
                ok := false
                break
            s += rawScore[i]
        score := ok ? s / smoothLen : na
    var bool  inRange = false
    var int   streak  = 0
    var int   dwell   = 0
    var float rLo     = na
    var float rHi     = na
    if not na(score)
        dwell += 1
        if inRange
            if not na(atrD) and (close > rHi + breakoutAtr * atrD or close < rLo - breakoutAtr * atrD)
                inRange := false
                streak  := 0
                dwell   := 0
            else
                streak := score <= exitRange ? streak + 1 : 0
                if streak >= confirmBars and dwell >= minDwell
                    inRange := false
                    streak  := 0
                    dwell   := 0
                else
                    rLo := math.min(rLo, low)
                    rHi := math.max(rHi, high)
        else
            streak := score >= enterRange ? streak + 1 : 0
            if streak >= confirmBars and dwell >= minDwell and not na(hhR)
                inRange := true
                streak  := 0
                dwell   := 0
                rLo     := llR
                rHi     := hhR
    [inRange, score]

[rngLraw, scoreL] = rangeDetector(smoothL, confL, dwellL)
[rngSraw, scoreS] = rangeDetector(smoothS, confS, dwellS)

// wielkość pozycji: docelowa ekspozycja × kapitał / cena, zaokrąglona W DÓŁ do kroku,
// ograniczona tak, by depozyt (wartość / dźwignia na giełdzie) zmieścił się w kapitale
posQty(float target) =>
    float raw    = target * strategy.equity / close
    float maxDep = strategy.equity * exchLev / close
    float q      = math.floor(math.min(raw, maxDep) / qtyStep + 1e-9) * qtyStep   // 1e-9: ochrona przed błędem zaokrąglenia
    q >= minQty and q > 0 ? q : 0.0

float qL = posQty(targetL)

goShort := goShort and qS > 0 and not goLong

    strategy.entry("SOL", strategy.long, qty = qL, comment = "Long SOL", alert_message = "Long SOL")

        strategy.exit("SL", "SOL", stop = stopPx, comment = "Stop long", alert_message = "Zamknij long SOL (stop loss)")

    lastQty := qS
    strategy.entry("SOLS", strategy.short, qty = qS, comment = "Short SOL", alert_message = "Short SOL")
    if useSLS

        strategy.exit("SLS", "SOLS", stop = stopPx, comment = "Stop short", alert_message = "Zamknij short SOL (stop loss)")
